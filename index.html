async function cargarCategorias() { const contenedor = document.getElementById("listaCategorias"); try { const respuesta = await fetch( SUPABASE_URL + "/rest/v1/CATEGOR%C3%8DAS?select=*&order=id.asc", { method: "GET", headers: { "apikey": SUPABASE_KEY, "Authorization": "Bearer " + SUPABASE_KEY } } ); if (!respuesta.ok) { throw new Error(await respuesta.text()); } const categorias = await respuesta.json(); console.log("Categorías recibidas:", categorias); contenedor.innerHTML = ""; /* TODOS LOS PRODUCTOS */ const botonTodos = document.createElement("button"); botonTodos.type = "button"; botonTodos.className = "categoria categoria-todos activa"; botonTodos.innerHTML = "

✨ Todos los productos

"; botonTodos.onclick = function() { filtrarCategoria( "TODOS", botonTodos ); }; contenedor.appendChild(botonTodos); /* CATEGORÍAS */ categorias.forEach(categoria => { const boton = document.createElement("button"); boton.type = "button"; boton.className = "categoria"; /* EMOJI DE SUPABASE */ const emoji = categoria.Emoji && String(categoria.Emoji).trim() ? String(categoria.Emoji).trim() : "🍬"; boton.innerHTML = `

${emoji} ${categoria.Nombre}

`; boton.onclick = function() { filtrarCategoria( categoria.Nombre, boton ); }; contenedor.appendChild(boton); }); } catch (error) { console.error( "ERROR CARGANDO CATEGORÍAS:", error ); contenedor.innerHTML = `

❌ No se pudieron cargar las categorías.

`; } }
